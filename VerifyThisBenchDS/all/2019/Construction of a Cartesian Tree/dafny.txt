class TreeNode {
    var value: int
    var left: TreeNode?
    var right: TreeNode?
    var parent: TreeNode?
    
    constructor(val: int)
        ensures value == val
        ensures left == null && right == null && parent == null
    {
        value := val;
        left := null;
        right := null;
        parent := null;
    }
}

predicate ValidCartesianTree(root: TreeNode?, sequence: seq<int>)
    reads if root != null then {root} + root.left + root.right else {}
{
    if root == null then
        |sequence| == 0
    else
        // Root has minimum value in sequence
        (forall i :: 0 <= i < |sequence| ==> root.value <= sequence[i]) &&
        // Left subtree contains elements before root in sequence
        (exists k :: 0 <= k < |sequence| && sequence[k] == root.value &&
         ValidCartesianTree(root.left, sequence[0..k]) &&
         ValidCartesianTree(root.right, sequence[k+1..]))
}

method ConstructCartesianTree(sequence: array<int>) returns (root: TreeNode?)
    requires sequence.Length > 0
    requires forall i, j :: 0 <= i < j < sequence.Length ==> sequence[i] != sequence[j] // distinct values
    ensures ValidCartesianTree(root, sequence[..])
{
    if sequence.Length == 0 {
        return null;
    }
    
    // Find minimum element
    var minIdx := 0;
    for i := 1 to sequence.Length
        invariant 0 <= minIdx < i
        invariant forall j :: 0 <= j < i ==> sequence[minIdx] <= sequence[j]
    {
        if sequence[i] < sequence[minIdx] {
            minIdx := i;
        }
    }
    
    root := new TreeNode(sequence[minIdx]);
    
    if minIdx > 0 {
        var leftSubarray := sequence[0..minIdx];
        root.left := ConstructCartesianTree(leftSubarray);
        if root.left != null {
            root.left.parent := root;
        }
    }
    
    if minIdx < sequence.Length - 1 {
        var rightSubarray := sequence[minIdx+1..];
        root.right := ConstructCartesianTree(rightSubarray);
        if root.right != null {
            root.right.parent := root;
        }
    }
}
