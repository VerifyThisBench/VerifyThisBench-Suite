class Node {
    int value;
    Node next;
    Node prev;
    Node nested_next;
    Node nested_prev;
    
    //@ resource state();
    
    Node(int v) {
        value = v;
        next = null;
        prev = null;
        nested_next = this;
        nested_prev = this;
    }
}
